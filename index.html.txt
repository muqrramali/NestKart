// CART
function addCart(id,pack,price){
  const p=prods.find(x=>x.id===id);if(!p)return;
  const key=`${id}_${pack}`;
  const ex=cart.find(x=>x.key===key);
  if(ex)ex.qty++;
  else cart.push({key,id,pack,price,name:p.name,img:p.images&&p.images[0]?p.images[0]:'',cat:p.cat,qty:1});
  saveC();updateCartCounts();renderCart();
}
function updateCartCounts(){
  const n=cart.reduce((a,x)=>a+x.qty,0);
  document.getElementById('cartN').textContent=n;
  const el2=document.getElementById('cartN2');
  if(el2)el2.textContent=n;
}
function openCart(){document.getElementById('cartOv').classList.add('open');document.getElementById('cartSb').classList.add('open');renderCart();}
function closeCart(){document.getElementById('cartOv').classList.remove('open');document.getElementById('cartSb').classList.remove('open');}
function renderCart(){
  const ci=document.getElementById('cartItems'),cf=document.getElementById('cartFoot');
  if(!cart.length){ci.innerHTML='<div class="empty-c"><span>🛒</span><p>Cart khali hai</p></div>';cf.innerHTML='';return;}
  ci.innerHTML=cart.map(item=>`
    <div class="citem">
      <div class="citem-img">${item.img?`<img src="${item.img}">`:(EM[item.cat]||'📦')}</div>
      <div class="citem-info">
        <div class="citem-name">${item.name} (Pack of ${item.pack})</div>
        <div class="citem-price">${pkrFmt(item.price)}</div>
        <div class="qty-row">
          <button class="qbtn" onclick="chgQty('${item.key}',-1)">−</button>
          <span class="qnum">${item.qty}</span>
          <button class="qbtn" onclick="chgQty('${item.key}',1)">+</button>
          <button class="rm-btn" onclick="rmCart('${item.key}')">🗑️ Remove</button>
        </div>
      </div>
    </div>`).join('');
  const total=cart.reduce((a,x)=>a+x.price*x.qty,0);
  cf.innerHTML=`<div class="cart-tot">
    <div class="ct-row"><span>Items (${cart.reduce((a,x)=>a+x.qty,0)})</span><span>${pkrFmt(total)}</span></div>
    <div class="ct-row"><span>Delivery</span><span style="color:#16a34a;">FREE</span></div>
    <div class="ct-row bold"><span>Total (COD)</span><span>${pkrFmt(total)}</span></div>
  </div>
  <button class="checkout-btn2" onclick="cartCheckout()">🛒 Checkout — Cash on Delivery</button>`;
}
function chgQty(key,d){const i=cart.find(x=>x.key===key);if(!i)return;i.qty+=d;if(i.qty<=0)cart=cart.filter(x=>x.key!==key);saveC();updateCartCounts();renderCart();}
function rmCart(key){cart=cart.filter(x=>x.key!==key);saveC();updateCartCounts();renderCart();}
function cartCheckout(){
  if(!cart.length)return;
  closeCart();
  // open order form for first cart item
  const first=cart[0];
  openOrder(first.id,first.pack,first.price);
}

// ADD FORM
function initAddForm(){
  prodImgs=[];descImgs=[];
  renderImgSlots();renderDescImgSlots();
}

function renderImgSlots(){
  const wrap=document.getElementById('imgsGrid');
  wrap.innerHTML='';
  for(let i=0;i<8;i++){
    const slot=document.createElement('div');
    slot.className='img-slot';
    if(prodImgs[i]){
      slot.innerHTML=`<img src="${prodImgs[i]}"><button class="rm-img" onclick="rmProdImg(event,${i})">✕</button>`;
    } else {
      slot.innerHTML=`<span style="font-size:24px;color:#ccc;">📷</span>`;
      slot.onclick=()=>triggerImgUpload(i);
    }
    wrap.appendChild(slot);
  }
}

function triggerImgUpload(idx){
  const inp=document.createElement('input');
  inp.type='file';inp.accept='image/*';
  inp.onchange=e=>{
    const f=e.target.files[0];if(!f)return;
    const r=new FileReader();
    r.onload=ev=>{prodImgs[idx]=ev.target.result;renderImgSlots();};
    r.readAsDataURL(f);
  };
  inp.click();
}

function rmProdImg(e,idx){e.stopPropagation();prodImgs[idx]=null;renderImgSlots();}

function renderDescImgSlots(){
  const wrap=document.getElementById('descImgsWrap');
  wrap.innerHTML=descImgs.map((img,i)=>`
    <div style="position:relative;">
      <div class="desc-ibox"><img src="${img}"></div>
      <button onclick="rmDescImg(${i})" style="position:absolute;top:-5px;right:-5px;background:#ef4444;color:white;border:none;border-radius:50%;width:18px;height:18px;font-size:10px;cursor:pointer;">✕</button>
    </div>`).join('');
  if(descImgs.length<6){
    wrap.innerHTML+=`<div class="desc-ibox" onclick="addDescImg()"><span>+</span></div>`;
  }
}

function addDescImg(){
  const inp=document.createElement('input');
  inp.type='file';inp.accept='image/*';
  inp.onchange=e=>{
    const f=e.target.files[0];if(!f)return;
    const r=new FileReader();
    r.onload=ev=>{descImgs.push(ev.target.result);renderDescImgSlots();};
    r.readAsDataURL(f);
  };
  inp.click();
}
function rmDescImg(i){descImgs.splice(i,1);renderDescImgSlots();}

function saveProd(){
  const name=document.getElementById('fn').value.trim();

  if(!name||!p1){toast2('❌ Name aur Pack 1 price zaroori hai!','e');return;}

  const imgs=prodImgs.filter(Boolean);
  const obj={

    descImgs:[...descImgs],
    images:imgs,
    p1:parseInt(p1),
    p2:document.getElementById('fp2').value?parseInt(document.getElementById('fp2').value):null,
    p3:document.getElementById('fp3').value?parseInt(document.getElementById('fp3').value):null,
